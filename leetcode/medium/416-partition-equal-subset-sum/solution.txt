        int sum=0;
        for(int i=0; i<n; i++) sum += nums[i];
        if (sum%2 != 0) return false;
        //else return false;
        int target = sum/2;
        //Set<Integer> set = new HashSet<>();
        boolean[] dp = new boolean[target+1];
        dp[0]=true;
        //Arrays.sort(nums);
        //set.add(0);
        for(int i=0; i<n; i++) {
            for (int j=target-nums[i]; j>=0; j--) {
                if (dp[j]) {
                        dp[j+nums[i]]=true;
                }
            }
        }
        if (dp[target]) return true;        
        return false;
    }
