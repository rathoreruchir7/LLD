class Solution {
    public long maxAlternatingSum(int[] nums) {
        int n = nums.length;
        long sum = 0;
        for(int i=0; i<n; i++) {
            nums[i] = Math.abs(nums[i]);
        }
        Arrays.sort(nums);
        for (int i=0; i<n/2; i++) {
            sum = sum - (nums[i]*nums[i]) + nums[n-1-i]*nums[n-1-i];
        }
        if (n%2 != 0) {
            sum += nums[n/2]*nums[n/2];
        }
        return sum;
    }
}
